fun isDivisible(n, x) {
    while (n >= x) {
        n = n - x;
    }
    return n == 0;
}

fun isPrime(n) {
    if (n <= 1) return false;
    var i = 2;
    while (i * i <= n) {
       if (isDivisible(n, i)) {
	  return false;
	}
	i = i + 1;
    }
      return true;
}
